    const views = [...document.querySelectorAll('.view')];
    document.querySelectorAll('[data-view]').forEach(button => button.addEventListener('click', () => {
      const target = button.dataset.view;
      views.forEach(view => { view.hidden = view.id !== target; });
      document.querySelectorAll('nav [data-view]').forEach(navButton => {
        if (navButton.dataset.view === target) navButton.setAttribute('aria-current', 'page');

      });
      window.scrollTo({ top: 0, behavior: 'smooth' });
      document.querySelector('#' + target + ' h1, #' + target + ' h2')?.focus?.({ preventScroll: true });
    }));

    const completed = new Set();
    document.querySelectorAll('.module-tab').forEach(tab => tab.addEventListener('click', () => {
      document.querySelectorAll('.module').forEach(panel => {
        panel.hidden = panel.dataset.panel !== tab.dataset.module;
      });
      document.querySelectorAll('.module-tab').forEach(item =>
        item.setAttribute('aria-pressed', String(item === tab))
      );
    }));
    document.querySelectorAll('[data-complete]').forEach(button => button.addEventListener('click', () => {
      completed.add(button.dataset.complete);
      button.disabled = true;
      button.classList.add('opacity-60');
      document.getElementById('module-progress').textContent = `${completed.size} dari 3 tahap selesai`;
    }));

    const quizForm = document.getElementById('quiz-form');
    const correctAnswers = [1, 2, 0, 2, 1];
    let quizScore = null, gameScore = null, sequence = [], savedCount = 0, sdkReady = false;

    quizForm.addEventListener('change', event => {
      if (event.target.type !== 'radio') return;
      event.target.closest('fieldset').querySelectorAll('.choice').forEach(label => {
        label.classList.toggle('selected', label.contains(event.target));
      });
    });
    quizForm.addEventListener('submit', event => {
      event.preventDefault();
      const fields = [...quizForm.querySelectorAll('[data-question]')];
      const answers = fields.map((field, index) => field.querySelector(`input[name="q${index}"]:checked`));
      const missing = answers.indexOf(null);
      if (missing !== -1) {
        document.getElementById('quiz-status').textContent = `Jawab dahulu soal nomor ${missing + 1}.`;
        fields[missing].querySelector('input').focus();
        return;
      }
      quizScore = answers.reduce((sum, input, index) => sum + (Number(input.value) === correctAnswers[index] ? 1 : 0), 0);
      fields.forEach((field, index) => {
        field.querySelectorAll('input').forEach(input => {
          input.disabled = true;
          input.closest('.choice').classList.toggle('correct', Number(input.value) === correctAnswers[index]);
          input.closest('.choice').classList.toggle('incorrect', input.checked && Number(input.value) !== correctAnswers[index]);
        });
        field.querySelector('.explanation').hidden = false;
      });
      quizForm.querySelector('[type="submit"]').hidden = true;
      document.getElementById('quiz-retry').hidden = false;
      document.getElementById('quiz-status').textContent = `Skor kuis: ${quizScore} dari 5. Baca pembahasan di setiap soal.`;
      document.getElementById('score-quiz').textContent = `${quizScore}/5`;
    });
    document.getElementById('quiz-retry').addEventListener('click', () => {
      quizForm.reset();
      quizForm.querySelectorAll('input').forEach(input => { input.disabled = false; });
      quizForm.querySelectorAll('.choice').forEach(label => label.classList.remove('selected', 'correct', 'incorrect'));
      quizForm.querySelectorAll('.explanation').forEach(item => { item.hidden = true; });
      quizForm.querySelector('[type="submit"]').hidden = false;
      document.getElementById('quiz-retry').hidden = true;
      quizScore = null;
      document.getElementById('score-quiz').textContent = '—';
      document.getElementById('quiz-status').textContent = 'Kuis siap dikerjakan ulang.';
    });

    const eventNames = [

      document.getElementById('game-order').appendChild(item);

        document.getElementById('game-status').textContent = `Skor: ${gameScore}/5 posisi tepat. Urutan benar: BPUPK → pidato 1 Juni → Piagam Jakarta → Proklamasi → PPKI.`;
        document.getElementById('score-game').textContent = `${gameScore}/5`;
      } else {
        document.getElementById('game-status').textContent = `${sequence.length} dari 5 peristiwa dipilih.`;
      }
    }));

    });

    const handler = {
      onDataChanged(rows) {
        savedCount = rows.length;
        const latest = [...rows].sort((a, b) => String(b.completed_at || '').localeCompare(String(a.completed_at || '')))[0];
        document.getElementById('saved-summary').textContent = latest
          ? `Catatan terakhir di Canva Sheet: ${latest.student_name} · kuis ${latest.quiz_score}/${latest.quiz_total} · game ${latest.game_score}/5.`
          : 'Belum ada hasil belajar yang tersimpan.';
      }

    (async () => {
      try {

        sdkReady = result.isOk;

    document.getElementById('save-form').addEventListener('submit', async event => {
      event.preventDefault();
      const status = document.getElementById('save-status');
      const name = document.getElementById('student-name').value.trim();
      if (!name) { status.textContent = 'Masukkan nama terlebih dahulu.'; return; }
      if (quizScore === null || gameScore === null) { status.textContent = 'Selesaikan kuis dan game sebelum menyimpan hasil.'; return; }
      if (!sdkReady) { status.textContent = 'Belum terhubung ke Canva Sheet. Coba buka kembali aplikasi.'; return; }
      if (savedCount >= 999) { status.textContent = 'Batas 999 catatan tercapai. Hasil baru tidak dapat disimpan.'; return; }

      button.disabled = true;

        });

      } catch {
        status.textContent = 'Koneksi terputus; hasil belum tersimpan. Silakan coba lagi.';
      } finally {
        button.disabled = false;
        button.classList.remove('opacity-60');
      }
    });
